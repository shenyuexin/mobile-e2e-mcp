### Wave 1: WDA Backend Edge Case Tests

<task type="test" wave="1">
  <name>Add edge case tests for WdaRealDeviceBackend</name>
  <files>packages/adapter-maestro/test/ios-backend-wda.test.ts (extend)</files>
  <action>
    Add these test cases to existing ios-backend-wda.test.ts:

    1. transformWdaSource with undefined fields:
    ```typescript
    test("transformWdaSource handles undefined optional fields", () => {
      const backend = new WdaRealDeviceBackend();
      const wdaSource = {
        type: "XCUIElementTypeButton",
        children: [],
      };
      const transformed = backend.transformWdaSource(wdaSource as any);
      assert.equal(transformed.AXLabel, null);
      assert.equal(transformed.title, null);
      assert.equal(transformed.AXValue, null);
      assert.equal(transformed.frame, undefined);
      assert.equal(transformed.enabled, true);
      assert.deepEqual(transformed.custom_actions, ["default"]);
    });
    ```

    2. transformWdaSource with deeply nested children:
    ```typescript
    test("transformWdaSource handles deeply nested children (3 levels)", () => {
      const backend = new WdaRealDeviceBackend();
      const wdaSource = {
        type: "XCUIElementTypeWindow",
        children: [
          {
            type: "XCUIElementTypeCell",
            children: [
              { type: "XCUIElementTypeButton", name: "Tap me", children: [] },
            ],
          },
        ],
      };
      const transformed = backend.transformWdaSource(wdaSource as any);
      assert.equal(transformed.children.length, 1);
      assert.equal(transformed.children[0].children.length, 1);
      assert.equal(transformed.children[0].children[0].type, "Button");
      assert.equal(transformed.children[0].children[0].AXLabel, "Tap me");
    });
    ```

    3. isClickableType edge cases:
    ```typescript
    test("isClickableType correctly identifies clickable and non-clickable types", () => {
      const backend = new WdaRealDeviceBackend();
      // Clickable
      assert.ok((backend as any).isClickableType("XCUIElementTypeButton"));
      assert.ok((backend as any).isClickableType("Button"));
      assert.ok((backend as any).isClickableType("XCUIElementTypeLink"));
      assert.ok((backend as any).isClickableType("XCUIElementTypeSwitch"));
      // Not clickable
      assert.equal((backend as any).isClickableType("XCUIElementTypeStaticText"), false);
      assert.equal((backend as any).isClickableType("XCUIElementTypeImage"), false);
      assert.equal((backend as any).isClickableType(undefined), false);
      assert.equal((backend as any).isClickableType("UnknownType"), false);
    });
    ```

    4. executeWdaRequest handles network errors:
    ```typescript
    test("executeWdaRequest handles network errors gracefully", async () => {
      // This test will fail because WDA isn't running, verifying error handling
      const backend = new WdaRealDeviceBackend();
      const result = await backend.executeWdaRequest("device-123", "GET", "/nonexistent");
      assert.equal(result.success, false);
      assert.ok(result.error?.includes("failed") || result.error?.includes("rejected"));
    });
    ```
  </action>

    - All existing 14 WDA tests still pass
  </acceptance_criteria>
